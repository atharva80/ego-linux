Chunk(streamName, chunk, encoding) {
    if (chunk == null) {
      return;
    }

    if (typeof chunk === 'string') {
      appendText(streamName, chunk);
      return;
    }

    if (Buffer.isBuffer(chunk) || ArrayBuffer.isView(chunk)) {
      appendText(
          streamName,
          Buffer.from(chunk).toString(
              typeof encoding === 'string' ? encoding : undefined));
      return;
    }

    if (typeof chunk === 'object') {
      return;
    }
  }

  function appendError(error) {
    if (error == null) {
      return;
    }

    if (typeof error?.stack === 'string' && error.stack.length > 0) {
      // One chunk, so streamed output doesn't show the newline on its own.
      appendText(
          'stderr',
          error.stack.endsWith('\n') ? error.stack : `${error.stack}\n`);
      return;
    }

    appendChunk('stderr', String(error));
  }

  function handleFatalError(error) {
    appendError(error);
    process.exit(1);
  }

  function handleUnhandledRejection(reason) {
    handleFatalError(reason);
  }

  let fatalErrorHandlersInstalled = false;

  function installFatalErrorHandlers() {
    if (fatalErrorHandlersInstalled) {
      return;
    }
    process.prependListener('uncaughtException', handleFatalError);
    process.prependListener('unhandledRejection', handleUnhandledRejection);
    fatalErrorHandlersInstalled = true;
  }

  function suspendFatalErrorHandlersForRepl() {
    if (!fatalErrorHandlersInstalled) {
      return;
    }
    process.removeListener('uncaughtException', handleFatalError);
    process.removeListener('unhandledRejection', handleUnhandledRejection);
    fatalErrorHandlersInstalled = false;
  }

  function makeBufferStream(streamName) {
    return new Writable({
      decodeStrings: false,
      write(chunk, encoding, callback) {
        appendChunk(streamName, chunk, encoding);
        callback(null);
      }
    });
  }

  const stdout = makeBufferStream('stdout');
  const stderr = makeBufferStream('stderr');

  globalThis.__egoNodeOutputCapture = {
    clear() {
      buffer.length = 0;
    },

    take() {
      const output = buffer.join('');
      buffer.length = 0;
      return output;
    },

    // Streams all later output to `callback(streamName, text)`, where
    // `streamName` is 'stdout' or 'stderr'. take() stays empty afterwards.
    setSink(callback) {
      sink = callback;
    },
  };

  globalThis.__egoInstallFatalErrorHandlers = installFatalErrorHandlers;
  globalThis.__egoSuspendPreloadFatalErrorHandlersForRepl =
      suspendFatalErrorHandlersForRepl;

  ObjectDefineProperty(process, 'stdout', {
    __proto__: null,
    configurable: true,
    enumerable: true,
    get: () => stdout,
  });
  ObjectDefineProperty(process, 'stderr', {
    __proto__: null,
    configurable: true,
    enumerable: true,
    get: () => stderr,
  });
})();

function evaluateScript(filename, source, options = undefined) {
  const inputType = options?.inputType;
  const print = options?.print === true;
  const stripTypes = options?.stripTypes === true;

  if (inputType === 'module') {
    return evalModuleEntryPoint(source, print);
  }

  if (inputType === 'module-typescript' && stripTypes) {
    return parseAndEvalModuleTypeScript(source, print);
  }

  if (inputType === 'commonjs') {
    return evalScript(filename, source, false, print, false);
  }

  if (inputType === 'commonjs-typescript' && stripTypes) {
    return parseAndEvalCommonjsTypeScript(
        filename, source, false, print, false);
  }

  if (stripTypes) {
    return evalTypeScript(filename, source, false, print, false);
  }

  return evalScript(filename, source, false, print, false);
}
